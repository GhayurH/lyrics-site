بادشاہ حسینؑ حسینؑ، بادشاہ حسینؑ

۱
یہ دین پناہی کا شرف کِس کو ملا ہے
یہ کس کا لقب ہے جو سرِعرش لکھا ہے

۲
اجمیر کے خواجہ نے یہ سچ خوب کہا ہے
لاریب کہ شبیرؑ ہی کلمے کی بقاء ہے

۳
شاہ است حسینؑ، بادشاہ است حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ
سردار نہ داد، دست در دستِ یزید
حقا کہ بِنا، لا الہ است حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۴
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ
مولا بڑا کام کیا، دینِ خدا عام کیا
کر نہ سکا کوئی کبھی، تُو نے وہی کام کیا
++ وعدہ وفا، کر کے رہا، سر نہ جُھکا، حق پہ جیا
سب سے بڑا میرا خدا، کرتا رہا، اعلان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۵
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
عشق تیرا عشقِ خدا، نوکِ سناں تک جو گیا
عشق میں جو تُو نے دیا، کوئی بھی وہ دے نہ سکا
++ نورِ خدا، دستِ خدا، تیرے سوا، کون ہوا
کہنا پڑا، لکھنا پڑا، ہے یہ تیرا، احسان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۶
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
سرورِ ذیشان حسینؑ، عظمتِ انسان حسینؑ
بولتا قرآن حسینؑ، میں تیرے قربان حسینؑ
++ میرِ حرم، رب کی قسم، تیرا یہ غم، ہوگا نہ کم
فرشِ عزا، حق کی صدا، ہے یہ تیرا، فیضان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۷
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
عزمِ جوان، تشنہ لباں، بے وطناں، بے کفناں
تیرے لئے ہے یہ فغاں، عرش و زمیں نوحہ کناں
++ دردِ دلاں، تشنہ دہاں، خشک زباں، نوکِ سناں
تن سے جدا، سر تھا مگر، پڑھتا رہا، قرآن حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۸
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
ہر جاء ہے حکومت تیری ہر دل میں بسا ہے
ہر دل میں عزاء خانہ تیرا آج سجا ہے
قرآنِ مصائب میں تیرا نام لکھا ہے
ماتم کی صدا ہے کہ یہ تُو بول رہا ہے
++ عشقِ خدا، تیری عطاء، نام تیرا، زندہ رہا
کرب و بلا، کی ہے صدا، تو ہے بڑا انسان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۹
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
تو زندہ عزاداری تیری زندہ رہے گی
وہ قوم جو بیدار ہے غم تیرا کرے گی
آواز اذاں جو تیرے روضے سے اٹھے گی
واللہ خدائی تیری آواز سنے گی
++ شاہِ زمن، ہادیءِ من، نازِ چمن، شیریں سخن
شانِ نبیؐ، شانِ علیؑ، یعنی میری، جان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۱۰
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
جنتِ مَن کرب و بلا، راحتِ مَن کرب و بلا
سرور و ریحانِ عزا، دولتِ مَن کرب و بلا
++ ہم ہیں، سفیرانِ عزا، شاہِ ھُدیٰ، تم پہ سدا
جان کریں تم پہ فِدا، ہے یہ میرا، ارمان حسینؑ
شاہ حسینؑ، بادشاہ حسینؑ
دیں است حسینؑ، دیں پناہ است حسینؑ

۱۱
کرب و بلا، عشقِ خدا، عشق کا تُو، سلطان حسینؑ
