کاٹا گیا حسین کا مقتل میں جب گلا
بس کچھ ہی فاصلے سے بہن دیکھتی رہی

۱
سر تن سے ہو رہا تھا برادر کا جب دعا

۲
گھیرا تھا ظالموں نے شہ مشرقین کو
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
اپنے غازی کو صدا دے رہی تھی زینب
بیکس بہن تڑپتی رہی کچھ نہ بس چلا

۳
خنجر تلے حسین کی گردن کو دیکھ کر
اے بھای کہہ کے گرتی تھی دکھیا زمیں پر
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
وہ وقت ایک بہن پہ قیامت کا وقت تھا

۴
پیاری بہن حسین کی سر پیٹتی رہی
سوکھے گلے پہ بھای کے چلتی رہی چھری
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
زنجیر بیکسی نے کسی تھی کچھ اس طرح

۵
پردیس میں علی کی دلاری تھی بیقرار
روتی تھی زار زار تڑپتی تھی بار بار
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
ہر سمت سے حسین پہ ہوتی رہی جفا

۶
نرغے میں ظالموں کے تھی بیٹی بتول کی
کچھ بھی نہ کرنے پای نواسی رسول کی
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
بھای کا سر نگاہوں کے آ گے ہی کٹ گیا

۷
سینے پہ جب حسین کے شمر لعیں چڑھا
زینب نے سر کو پیٹ کے دریا کا رخ کیا
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
اس وقت تھر تھرانے لگی ارض کربلا

۸
زینب پہ کربلا میں جو کوہ علم گرا
ٹوٹے نہ وہ پہاڑ کسی پہ بھی اے خدا
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
بھای کو قتل کرتا رہا شمر بے حیا

۹
وایض نہ لکھ سکے گا کبھی بھی وہ غم کوئی
زینب کے دل پہ تیغ جو عاشور کو چلی
زینب بچانے پای نہ اپنے حسین کو
سوے دریا مڑ کر دیکھتی تھی زینب
بھای کا خون خاک میں ملتا چلا گیا
