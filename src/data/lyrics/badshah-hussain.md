بادشاہ حسین حسین

یہ دین پناہی کا شرف کس کو ملا ہے
یہ کس کا لقب ہے جو سر عرش لکھا ہے
اجمیر کے خواجہ نے یہ سچ خوب کہا ہے
لا ریب کے شبیر ہی کلمے کی بقاء ہے

شاہ است حُسین بادشاہ است حسین
دیں است حسین دیں پناہ است حسین
سرداد نہ داد دست در دست یزید
حقا کے بنا لا الٰہ است حسین

کربوبلا عشقِ خدا
عشق کا تو سلطان حسین
شاہ حُسین بادشاہ حسین
دیں است حسین دیں پناہ است حسین
شاہ حُسین بادشاہ حسین
دیں است حسین دیں پناہ است حسین

۱
مولا بڑا کام کیا
دین خدا آم کیا
کر نہ سکا کوئی کبھی
تو نے وہی کام کیا
وعدہ وفا کر کے رہا
سر نہ جھکا حق پہ جیا
سب سے بڑا میرا خدا
کرتا رہا اعلان حسین

۲
عشق تیرا عشقِ خدا
نوکِ سِناں تک جو گیا
عشق میں جو تو نے دیا
کوئی بھی وہ دے نہ سکا
نورِ خدا دست خدا
تیرے سوا کون ہوا
کہنا پڑا لکھنا پڑا
ہے یہ تیرا احسان حسین

۳
سرور ذیشان حسین
عظمت انسان حسین
بولتا قرآن حسین
میں تیرے قربان حسین
میر حرم رب کی قسم
تیرا یہ غم ہوگا نہ کم
فرش عزا حق کی صدا
ہے یہ تیرا فیضان حسین

۴
عزمِ جواں تشنہ لباں
بے وطناں بے کفنا
تیرے لیےء ہے یہ فغاں
عرش و زمیں نوحہ کناں
درد دلاں تشنہ دہاں
خشک زباں نوکِ سِناں
تن سے جدا سر تھا مگر
پڑھتا رہا قرآن حسین

۵
ہر جا ہے حکومت تیری ہر دل میں بسا ہے
ہر دل میں عزاخانہ تیرا آج سجا ہے
قرآن مصائب میں تیرا نام لکھا ہے
ماتم کی صدا ہے کہ یہ تو بول رہا ہے
عشقِ خدا تیری ادا
نام تیرا زندہ رہا
کربوبلا کی ہے صدا
تو ہے بڑا انسان حسین

۶
تو زندہ عزاداری تیری زندہ رہے گی
وہ قوم جو بیدار ہے غم تیرا کرے گی
آواز اذاں جو تیرے روضے سے اٹھے گی
واللہ خدای تیری آواز سنے گی
شاہ زمن ہادیء من
ناز چمن شیریں سخن
شان نبی شان علی
یعنی میری جان حسین

۷
جنت من کربوبلا
راحتِ من کربوبلا
سرور و ریحان عزا
دولتِ من کربوبلا
ہم ہیں سفیران عزا
شاہ ہدا تم پہ صدا
جان کریں تم پہ فدا
ہے یہ میرا ارمان حسین
