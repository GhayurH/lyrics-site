یوں رسن بستہ نہ ہوتی کبھی بنت زہرا
میرا عباس جو ہوتا میرا عباس جو ہوتا

کون بلوے میں پھراتا مجھے یوں بے پردہ
میرا عباس جو ہوتا میرا عباس جو ہوتا

۱
آج تک میں نے سنا بھی نہ تھا بازار کا نام
ہاں میری ماں نے لیا تھا کبھی دربار کا نام
میں کہاں اور کہاں اہل جفا کا بلوہ

۲
کون ظالم میرے خیموں کو جلاتا آ کر
کون بیمار کو بستر سے اٹھاتا آ کر
کہاں جل سکتا تھا بے شیر کا ننھا جھولا

۳
آج کوفے میں نہ ہوتی میری یہ بربادی
جہاں قیدی ہوں کہی جاتی وہی شہزادی
شانے کٹوا کے نہ سوتا جو میرا شہزادہ

۴
اس طرح آل نبی جاتی نہ درباروں میں
نام زینب کا نہ لیتا کوئی بازاروں میں
خانۂ آیۂ تطہیر نہ ہوتا رسوا

۵
مارتا ہے جو سکینہ کو طمانچے کوئی
پشت عابد پہ لگا جاتا ہے درے کوئی
نظر آتا نہ یہاں کوئی ستمگر زندہ

۶
گود میں ماں کی ہمکتا علی اصغر میرا
کھینچتا آ کے قناتیں علی اکبر میرا
رات بھر وہ میرے خیمے کا تلایا پھرتا

۷
میرے بچوں کو اگر موت نہیں آ جاتی
اپنے گھر میں بھی کبھی چاند سی بہوئیں لاتی
باندھتی عون و محمد کے سروں پر سہرا

۸
میں ہوں زینب میرے بھائی تھے حمیت والے
حوصلے والے وفا والے شجاعت والے
دیکھ سکتا نہ تھا سورج کبھی میرا چہرہ

۹
تھا بہت عامر و نیر یہ بہن کو بھی خیال
کہاں بے پردگیٔ اہل حرم کا تھا سوال
خود ہی ہو جاتا ردا بڑھ کے علم کا سایہ
