ہائے سكینہ ہائے ہائے ہائے شام

۱
دن رات صدا جو آتی ہے
ہر دل کو جو تڑپاتی ہے
یہ بالی سكینہ ہے
زنداں میں جو روتی ہے
بابا کو بلاتی ہے
یہ بالی سكینہ ہے
دن رات صدا جو آتی ہے
ہر دل کو جو تڑپاتی ہے
یہ بالی سكینہ ہے
نازوں سے پلی ہے یہ بچی
سب اس کے ناز اُٹھاتے تھے
کبھی ماں کبھی بابا بھائی چچا
سب اس پہ واری جاتے تھے
کبھی یہ گودی کبھی وہ گودی
معصومہ کو یاد آتی ہے
یہ بالی سكینہ ہے
زنداں میں جو روتی ہے
بابا کو بلاتی ہے
یہ بالی سكینہ ہے
دن رات صدا جو آتی ہے
ہر دل کو جو تڑپاتی ہے
یہ بالی سكینہ ہے
بابا سے بچھڑ کر بیٹی نے
کیا کیا نہ مصائب دیکھے ہیں
پیاسی بھی رہی قیدی بھی بنی
ظالم سے طمانچے کھائے ہیں
جو سینۂ شاہ پہ سوتی تھی
اب وہ ہی خاک پہ سوتی ہے
یہ بالی سكینہ ہے
زنداں میں جو روتی ہے
بابا کو بلاتی ہے
یہ بالی سكینہ ہے

۲
زندان کے در سے جو گزرے
بی‌بی ہر ایک سے کہتی ہے
بابا سے ملو تو کہہ دینا
تمہیں یاد سكینہ کرتی ہے
تم پاس چلے آؤ بابا
رو رو کے وہ چلاتی ہے
یہ بالی سكینہ ہے
زنداں میں جو روتی ہے
بابا کو بلاتی ہے
یہ بالی سكینہ ہے
دن رات صدا جو آتی ہے
ہر دل کو جو تڑپاتی ہے
یہ بالی سكینہ ہے
عاشور سے لے کر زنداں تک
ہر ظلم اسے یاد آتا ہے
عباس اکبر قاسم اصغر
بابا کا غم تڑپاتا ہے
زندان میں ماتم کر کر کے
جو سب کا سوگ مناتی ہے
یہ بالی سكینہ ہے
زنداں میں جو روتی ہے
بابا کو بلاتی ہے
یہ بالی سكینہ ہے

۳
سجاد سے رو رو کہتی ہے
کب ملنے بابا آئیں گے
اچھے بھیا یہ بتلا دو
کیا ہم بھی کبھی گھر جائیں گے
کبھی بھائی سے کبھی اماں سے
ہر وقت سوال جو کرتی ہے
یہ بالی سکینہ ہے

بابا کو بلاتی ہے
یہ بالی سكینہ ہے
